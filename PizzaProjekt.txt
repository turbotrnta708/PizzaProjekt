<script>
function calculateTotal() {
  let pizza = document.getElementById("pizza");
  let size = document.querySelector('input[name="size"]:checked');
  let delivery = document.querySelector('input[name="delivery"]:checked');
 
  let total = Number(pizza.value) + Number(size.value) + Number(delivery.value);
  let toppings = "";
 
  // Add every checked topping; its label text is used as the name
  document.querySelectorAll(".topping:checked").forEach(function(t) {
    total += Number(t.value);
    toppings += "\n   • " + t.parentElement.textContent.trim();
  });
 
  document.getElementById("total").textContent = "Gesamt: €" + total.toFixed(2);
 
  document.getElementById("summary").textContent =
    "Name: " + document.getElementById("fname").value + " " + document.getElementById("lname").value + "\n" +
    "Pizza: " + pizza.options[pizza.selectedIndex].text + "\n" +
    "Größe: " + size.parentElement.textContent.trim() + "\n" +
    "Beläge:" + (toppings || " keine") + "\n" +
    "Lieferart: " + delivery.parentElement.textContent.trim() + "\n" +
    "Gesamt: €" + total.toFixed(2);
}
 
// The reset button resets the form fields itself; this clears the result
function clearResult() {
  document.getElementById("total").textContent = "Gesamt: €0.00";
  document.getElementById("summary").textContent = "";
}
</script>
